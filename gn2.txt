* 1. Data Declarations and Object Instantiation
DATA(lo_excel)     = NEW zcl_excel( ).
DATA(lo_worksheet) = lo_excel->get_active_worksheet( ).

* 2. Define Styles (Row 7)
" Style 1: Previous Gross (Background: Red, Font: Red)
DATA(lo_style_prev) = lo_excel->add_new_style( ).
lo_style_prev->fill->filltype = zcl_excel_style_fill=>c_fill_solid.
lo_style_prev->fill->fgcolor-rgb = zcl_excel_style_color=>c_red.
lo_style_prev->font->color-rgb   = zcl_excel_style_color=>c_red. " Note: text will blend with background
DATA(lv_guid_prev) = lo_style_prev->get_guid( ).

" Style 2: Current Gross (Background: White, Font: Black)
DATA(lo_style_curr) = lo_excel->add_new_style( ).
lo_style_curr->fill->filltype = zcl_excel_style_fill=>c_fill_solid.
lo_style_curr->fill->fgcolor-rgb = zcl_excel_style_color=>c_white.
lo_style_curr->font->color-rgb   = zcl_excel_style_color=>c_black.
DATA(lv_guid_curr) = lo_style_curr->get_guid( ).

" Style 3: Deviation (Background: Yellow, Font: Black)
DATA(lo_style_dev) = lo_excel->add_new_style( ).
lo_style_dev->fill->filltype = zcl_excel_style_fill=>c_fill_solid.
lo_style_dev->fill->fgcolor-rgb = zcl_excel_style_color=>c_yellow.
lo_style_dev->font->color-rgb   = zcl_excel_style_color=>c_black.
DATA(lv_guid_dev) = lo_style_dev->get_guid( ).

* 3. Define Coordinates
DATA: lv_col TYPE zexcel_cell_column VALUE 7. " Starts at 7, which is Column 'G'
CONSTANTS: lc_row_wt  TYPE zexcel_cell_row VALUE 6,
           lc_row_txt TYPE zexcel_cell_row VALUE 7.

* 4. Fetch Unique Wage Types from your table
* (Assuming the field name for Wage Type is 'LGART'. Adjust if it differs).
SELECT DISTINCT lgart 
  FROM zespy_pmi_g2n_wt 
  INTO TABLE @DATA(lt_wage_types)
  ORDER BY lgart.

* 5. Populate Cells Dynamically
LOOP AT lt_wage_types INTO DATA(ls_wt).

  " --- 1st Column in set (e.g., G) ---
  lo_worksheet->set_cell( ip_column = lv_col ip_row = lc_row_wt  ip_value = ls_wt-lgart ).
  lo_worksheet->set_cell( ip_column = lv_col ip_row = lc_row_txt ip_value = 'Previous Gross - WTs detail' ip_style = lv_guid_prev ).
  lv_col = lv_col + 1. " Advance to next column

  " --- 2nd Column in set (e.g., H) ---
  lo_worksheet->set_cell( ip_column = lv_col ip_row = lc_row_wt  ip_value = ls_wt-lgart ).
  lo_worksheet->set_cell( ip_column = lv_col ip_row = lc_row_txt ip_value = 'Current Gross - WTs detail' ip_style = lv_guid_curr ).
  lv_col = lv_col + 1. " Advance to next column

  " --- 3rd Column in set (e.g., I) ---
  lo_worksheet->set_cell( ip_column = lv_col ip_row = lc_row_wt  ip_value = ls_wt-lgart ).
  lo_worksheet->set_cell( ip_column = lv_col ip_row = lc_row_txt ip_value = 'Deviation Current Vs Previous' ip_style = lv_guid_dev ).
  lv_col = lv_col + 1. " Advance to next column for the next Wage Type group

ENDLOOP.
